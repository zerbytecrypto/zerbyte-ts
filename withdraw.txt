<script>

    // =====================================================
    // SUPABASE CONFIG
    // =====================================================

    const supabaseUrl =
        'https://odvjvyctqaidcexrqmrc.supabase.co';

    const supabaseAnonKey =
        'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6Im9kdmp2eWN0cWFpZGNleHJxbXJjIiwicm9sZSI6ImFub24iLCJpYXQiOjE3Njc5MTU5ODEsImV4cCI6MjA4MzQ5MTk4MX0.anicfODTH70oAhpniUMI_9KWQL7hie6i9cZHzUJa6hU';

    const supabaseClient =
        supabase.createClient(
            supabaseUrl,
            supabaseAnonKey
        );


    // =====================================================
    // VARIABLES
    // =====================================================

    let currentUser = null;
    let currentBalance = 0;



    // =====================================================
    // INITIALIZE PAGE
    // =====================================================

    async function initWithdrawPage() {

        try {

            // Get currently logged-in Supabase user
            const {
                data: { user },
                error: authError
            } = await supabaseClient.auth.getUser();


            if (authError) {
                throw authError;
            }


            // User is NOT logged in
            if (!user) {

                window.location.href = 'auth.html';

                return;
            }


            currentUser = user;


            // Display email
            document.getElementById('user-email').innerText =
                user.email || '';


            // =================================================
            // GET USER PROFILE
            // =================================================

            const {
                data: profile,
                error: profileError
            } = await supabaseClient
                .from('profiles')
                .select('balance')
                .eq('id', user.id)
                .single();


            if (profileError) {
                throw profileError;
            }


            if (!profile) {

                throw new Error(
                    'Your Zerbyte profile could not be found.'
                );

            }


            currentBalance =
                Number(profile.balance || 0);


            displayBalance();

        } catch (error) {

            console.error(
                'Withdrawal initialization error:',
                error
            );

            alert(
                'Unable to load your account. Please log in again.'
            );

            window.location.href = 'auth.html';

        }

    }



    // =====================================================
    // DISPLAY BALANCE
    // =====================================================

    function displayBalance() {

        document.getElementById('balance').innerText =
            '$' + currentBalance.toLocaleString(
                undefined,
                {
                    minimumFractionDigits: 2,
                    maximumFractionDigits: 2
                }
            );

    }



    // =====================================================
    // WITHDRAWAL FORM
    // =====================================================

    document
        .getElementById('withdraw-form')
        .addEventListener(
            'submit',
            async function(event) {

                event.preventDefault();


                const button =
                    document.getElementById('withdraw-btn');

                const amountError =
                    document.getElementById('amount-error');


                amountError.classList.add('hidden');


                // -------------------------------------------------
                // Get amount
                // -------------------------------------------------

                const amount =
                    Number(
                        document
                            .getElementById('withdraw-amount')
                            .value
                    );


                // -------------------------------------------------
                // Validate amount
                // -------------------------------------------------

                if (
                    !Number.isFinite(amount) ||
                    amount <= 0
                ) {

                    amountError.innerText =
                        'Enter a valid withdrawal amount.';

                    amountError.classList.remove('hidden');

                    return;
                }


                if (amount > currentBalance) {

                    amountError.innerText =
                        'Withdrawal amount exceeds your available balance.';

                    amountError.classList.remove('hidden');

                    return;
                }


                // -------------------------------------------------
                // Get bank details
                // -------------------------------------------------

                const accountName =
                    document
                        .getElementById('account-name')
                        .value
                        .trim();

                const accountNumber =
                    document
                        .getElementById('account-number')
                        .value
                        .trim();

                const bankName =
                    document
                        .getElementById('bank-name')
                        .value
                        .trim();

                const bankCode =
                    document
                        .getElementById('bank-code')
                        .value
                        .trim();

                const phone =
                    document
                        .getElementById('phone')
                        .value
                        .trim();


                // -------------------------------------------------
                // Disable button
                // -------------------------------------------------

                button.disabled = true;

                button.classList.add(
                    'opacity-50',
                    'cursor-not-allowed'
                );

                button.innerHTML =
                    '<i class="fa-solid fa-spinner fa-spin mr-2"></i> Processing...';


                try {

                    // -------------------------------------------------
                    // Send request to secure PostgreSQL function
                    // -------------------------------------------------

                    const {
                        data,
                        error
                    } = await supabaseClient.rpc(
                        'create_withdrawal',
                        {
                            withdrawal_amount: amount,

                            withdrawal_details: {

                                account_name: accountName,

                                account_number: accountNumber,

                                bank_name: bankName,

                                bank_code: bankCode,

                                phone: phone

                            }

                        }
                    );


                    if (error) {

                        throw error;

                    }


                    // -------------------------------------------------
                    // Successful withdrawal
                    // -------------------------------------------------

                    if (!data || data.success !== true) {

                        throw new Error(
                            'Withdrawal could not be completed.'
                        );

                    }


                    // Update local balance
                    currentBalance =
                        Number(data.new_balance || 0);


                    displayBalance();


                    // Show amount in success modal
                    document.getElementById(
                        'success-amount'
                    ).innerText =
                        '$' +
                        amount.toLocaleString(
                            undefined,
                            {
                                minimumFractionDigits: 2,
                                maximumFractionDigits: 2
                            }
                        );


                    // Show success modal
                    document
                        .getElementById('success-modal')
                        .classList.remove('hidden');


                } catch (error) {

                    console.error(
                        'Withdrawal error:',
                        error
                    );


                    let message =
                        error.message ||
                        'Withdrawal failed. Please try again.';


                    // Cleaner messages for database errors
                    if (
                        message
                            .toLowerCase()
                            .includes('insufficient balance')
                    ) {

                        message =
                            'Insufficient balance.';

                    }


                    alert(message);


                    // Re-enable button
                    button.disabled = false;

                    button.classList.remove(
                        'opacity-50',
                        'cursor-not-allowed'
                    );

                    button.innerHTML =
                        '<i class="fa-solid fa-paper-plane mr-2"></i> Submit Withdrawal';

                }

            }
        );



    // =====================================================
    // START
    // =====================================================

    document.addEventListener(
        'DOMContentLoaded',
        initWithdrawPage
    );

</script>
